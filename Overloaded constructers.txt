public class User {

    String username;
    String email;
    int age;

    User(){
        this.username = "Guest";
        this.email = "Not provided";
        this.age = 0;
    }

    User(String name){
        this.username = name;
        this.email = "Not provided";
        this.age = 0;
    }

    User(String name, String email){
        this.username = name;
        this.email =  email;
        this.age = 0;
    }

    User(String name, String email, int age){
        this.username = name;
        this.email = email;
        this.age = age;
    }
}
