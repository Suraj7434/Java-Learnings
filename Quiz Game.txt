import java.sql.Array;
import java.sql.SQLOutput;
import java.util.Arrays;
import java.util.Scanner;

public class Main {

    static Scanner scanner = new Scanner(System.in);
    public static void main(String[] args){

        String[] questions = {"Which keyword is used to create an object in Java?",
                             "What is the output of System.out.println(5 + 3 * 2);?",
                             "Which loop is commonly used to iterate directly through every element of an array?",
                             "Which data type is used to store a single character in Java?",
                             "What does the return statement do inside a Java method?"};

        String[][] options = {{"New","Create","Class","Object"},
                              {"13","10","16","11"},
                              {"Try loop","If loop","Enhanced for loop","switch loop"},
                              {"String","Character","Char","Text"},
                              {"Stops loop permanently","Returns a value","repeats method","created new method"}};

        int[] answers = {1, 4, 3, 3, 2};
        int score = 0;

        for(int i = 0; i < questions.length; i++){
            System.out.println(questions[i]);

            for( String ques : options[i]){
                System.out.println(ques);
            }

            System.out.print("Enter your answer choice(1-4): ");
            int choice = scanner.nextInt();

            if(choice == answers[i]){
                System.out.println("correct!");
                score++;
            }
            else{
                System.out.println("Wrong!");
            }
            System.out.println();


        }
        System.out.println("Your total score is: "+ score + " " + " out of " + questions.length);
        if(score >= 3){
            System.out.println("Congratulations! You have passed the exam");
        }
        else{
            System.out.println("Opps! failed! try again next time");
        }


    }

}
